\title{The Connes--St{\o}rmer entropy of two maximal abelian subalgebras:\\
a closed form for $M_2$ and partitions that beat bases in $M_n$}
\author{Irina Nikolaeva \and Andrej Novikov}
\date{}

\begin{document}
\maketitle

\begin{abstract}
The Connes--St{\o}rmer relative entropy $H(P|Q)$ measures the relative position of two subalgebras $P,Q$ of a finite
von Neumann algebra. Its simplest non-trivial instance is a pair of maximal abelian subalgebras of $M_n(\C)$, the
diagonal algebra $\Delta_n$ and a rotated copy $U\Delta_nU^*$. Choda's value $h(U)$, the averaged row entropy of the
unistochastic matrix $(|u_{ij}|^2)$, is a lower bound, and Petz, Sz\'ant\'o and Weiner showed that it is not the
entropy already for $n=2$; a formula for $H(\Delta_2|U\Delta_2U^*)$ has been described as open by them and by Bakshi and
Guin. We prove that for $U\in U(2)$ with $|u_{11}|^2=\cos^2\theta$,
\[
  H(\Delta_2|U\Delta_2U^*)=\max_{\alpha\in\R}\int_{\alpha-\theta}^{\alpha}\sin(2\beta)\log\tan^2\beta\,d\beta,
\]
attained by an explicit orthonormal basis whose angle we locate. Consequently $H\le\kappa\min(\theta,\frac\pi2-\theta)$
with $\kappa=\max_\beta\sin(2\beta)\log\tan^2\beta\approx\CSKappa$, and $H/\theta\to\kappa$ as $\theta\to0$, whereas Choda's
value is $o(\theta)$. For arbitrary subalgebras of $M_n$ we note the bound $H\le G^*$ by the maximum of the entropy
difference over states, with equality exactly when $\frac1n$ is a convex combination of maximisers; it is attained for
$n=2$. For $n=3,4,5$ a batched gradient search over partitions of unity and over orthonormal bases, with
high-precision recomputation of the best points, shows a different picture: for a small fraction of Haar-random $U$
the best partition found, with $n+1$ rank-one elements, beats every orthonormal basis found, and in all samples the best
value found lies strictly between $h(U)$ and the computed $G^*$. We state these findings as observations and conjectures.
\end{abstract}

\noindent\textbf{Keywords:} Connes--St{\o}rmer entropy; relative entropy of subalgebras; maximal abelian
subalgebras; unistochastic matrices; partitions of unity; complementary bases; numerical optimisation.\\
\textbf{MSC 2020:} 46L37, 94A17; 46L55, 15B51.

\section{Introduction}\label{sec:intro}

Let $\M$ be a finite von Neumann algebra with a faithful normal tracial state $\tau$, let $\eta(t)=-t\log t$, and for a
von Neumann subalgebra $P\subset\M$ let $E_P$ denote the $\tau$-preserving conditional expectation onto $P$. The relative
entropy of Connes and St{\o}rmer~\cite{connes1975} of two subalgebras $P,Q$ is
\begin{equation}\label{eq:cs}
  H(P|Q)=\sup\sum_j\bigl[\tau\eta(E_Qx_j)-\tau\eta(E_Px_j)\bigr],
\end{equation}
the supremum being taken over all finite partitions of unity $x_j\ge0$, $\sum_jx_j=1$, in $\M$. Connes and St{\o}rmer
introduced it for finite-dimensional subalgebras on the way to the entropy of automorphisms~\cite{connes1975}; Pimsner
and Popa extended it to arbitrary subalgebras and related the entropy $H(\M|N)$ of an inclusion $N\subset\M$ to the
Jones index~\cite{pimsner1986,jones1983}. The partition $x_1=1$ shows $H(P|Q)\ge0$, and $H(P|P)=0$. As a measure of the
relative position of two subalgebras (cf.~\cite{bakshi}), its simplest non-trivial instance is a pair of maximal
abelian subalgebras (masas) of $M_n=M_n(\C)$ with $\tau=\Tr/n$: the diagonal algebra $\Delta_n$ and a rotated copy
$U\Delta_nU^*$, $U$ unitary.

The partition into the minimal projections $e_{jj}$ of $\Delta_n$ gives the lower bound
\begin{equation}\label{eq:choda}
  H(\Delta_n|U\Delta_nU^*)\ \ge\ h(U)=\frac1n\sum_{i,j}\eta\bigl(|u_{ij}|^2\bigr),
\end{equation}
the entropy of the unistochastic matrix $(|u_{ij}|^2)$, averaged over its rows. Choda~\cite{choda} showed that $h(U)$ is
the value of a modified relative entropy in which the partitions are taken in $\Delta_n$. The expectation that
$H(\Delta_n|U\Delta_nU^*)=h(U)$ was disproved for $n=2$ by Petz, Sz\'ant\'o and Weiner, who remarked that the correct
formula ``seems to be a hard problem''~\cite[Appendix]{petz2009}. Bakshi and Guin bound the Connes--St{\o}rmer entropy
of pairs of spin model subfactors built from $2\times2$ complex Hadamard matrices from below by $h(U)$ and note that a
formula for $H(\Delta_n|U\Delta_nU^*)$ is open even for $n=2$~\cite[Section~2.3 and Theorem~6.22]{bakshi}. At the other
extreme, for complementary masas ($|u_{ij}|^2=\frac1n$ for all $i,j$, e.g.\ the Fourier matrix) one has
$h(U)=\log n$, which is the largest possible value, and $H=\log n$ holds only for complementary
masas~\cite[Theorem~3]{petz2009} (see also~\cite[Theorem~4.11]{ghosh2026}).

\paragraph{Results.} Write $h(p)=\eta(p)+\eta(1-p)$ for the binary entropy and
\begin{equation}\label{eq:psi}
  \psi(\beta)=\sin(2\beta)\log\tan^2\beta\qquad(\beta\in\R),
\end{equation}
extended by continuity with $\psi=0$ on $\frac\pi2\Z$. Every unitary $U\in M_2$ satisfies $|u_{11}|=|u_{22}|=\cos\theta$ for
a unique $\theta\in[0,\frac\pi2]$.

\begin{thmA}
Let $U\in M_2$ be unitary, $c=|u_{11}|^2=\cos^2\theta$ with $\theta\in[0,\frac\pi2]$, and
\[
  \varphi(\alpha)=h\bigl(\cos^2(\alpha-\theta)\bigr)-h(\cos^2\alpha)=\int_{\alpha-\theta}^{\alpha}\psi(\beta)\,d\beta .
\]
Then $H(\Delta_2|U\Delta_2U^*)=\max_{\alpha\in\R}\varphi(\alpha)$. The maximum is attained by a partition into the two
projections onto an explicit orthonormal basis of $\C^2$, determined by a maximiser $\alpha^*$ of $\varphi$ and the phases
of $U$. For $\frac12<c<1$ the maximiser is unique modulo $\frac\pi2$, lies in $(\theta-\frac\pi4,0)$ and is the unique
solution there of $\psi(\alpha)=\psi(\alpha-\theta)$; $H$ is invariant under $c\mapsto1-c$
(Theorem~\ref{thm:n2} and Proposition~\ref{prop:angle}).
\end{thmA}

\begin{thmB}
(i) For unital $*$-subalgebras $P,Q\subset M_n$ let $G(\rho)=S(E_Q\rho)-S(E_P\rho)$ on the states $\rho$ of $M_n$, with the
von Neumann entropy $S$, and $G^*=\max_\rho G(\rho)$. Then $H(P|Q)\le G^*$, with equality if and only if $\frac1n$ is a
convex combination of maximisers of $G$ (Proposition~\ref{prop:envelope}). For two masas of $M_2$ equality holds.\\
(ii) For $n=2$, with the notation of Theorem~A, $h(c)\le H(\Delta_2|U\Delta_2U^*)\le\kappa\min(\theta,\frac\pi2-\theta)$, where
$\kappa=\max\psi=2\sqrt{1-y^2}/y\approx\CSKappa$ and $y\in(0,1)$ is the unique solution of $y\artanh y=1$. As $\theta\to0$,
\[
  H(\Delta_2|U\Delta_2U^*)=\kappa\theta-\frac{\kappa+4/\kappa}{6}\,\theta^3+O(\theta^4),\qquad h(c)=o(\theta),
\]
so $H/h(U)\to\infty$ as $c\to1$ (Theorem~\ref{thm:bound}).
\end{thmB}

For $n\ge3$ we have no closed form. We computed $H(\Delta_n|U\Delta_nU^*)$ for Haar-random $U$ with $n=3,4,5$ by
batched multistart gradient ascent over partitions of unity with $n^2$ elements and, separately, over orthonormal bases,
with the PyTorch library \texttt{torch\_\allowbreak vn\_\allowbreak algebra}~\cite{lib} (Section~\ref{sec:numerics}). The
search reproduces Theorem~A for $n=2$ to $\CSTwoDevLib$ and shows:

\begin{obsC}
For $n=3,4,5$ the supremum~\eqref{eq:cs} for two masas is in general not attained at an orthonormal basis: for
$\CSThreeFlagged/\CSThreeU$, $\CSFourFlagged/\CSFourU$ and $\CSFiveFlagged/\CSFiveU$ of the sampled unitaries a partition
with $n+1$ rank-one elements beats the best basis found, by up to $\CSThreeGapMax$ ($n=3$); the partition values are
recomputed with $\CSDigits$ digits at explicit points. In all samples the best value found exceeds $h(U)$ and stays below
the computed $G^*$: Choda's partition is not optimal, and the bound of Theorem~B(i) does not appear to be attained.
\end{obsC}

Since no upper bound over all bases is certified, Observation~C is not a theorem; we formulate the corresponding
conjectures in Section~\ref{sec:numerics} and list open problems in Section~\ref{sec:open}.

\paragraph{What is new.} The ensemble form of $H$ (Proposition~\ref{prop:psw}), the bound $H\le\log n$, the strict
inequality $H(\Delta_2|U\Delta_2U^*)>h(U)$ for $c\notin\{0,\frac12,1\}$ and the characterisation of $H=\log n$ are due to
Petz, Sz\'ant\'o and Weiner~\cite{petz2009}; the lower bound~\eqref{eq:choda} and its interpretation are Choda's~\cite{choda}.
Theorem~A, the bounds of Theorem~B and Observation~C are new. Theorem~B(i) is an elementary consequence of the ensemble form;
we state it because its equality case separates $n=2$ from $n\ge3$. A short account of part of these results appears as a
use case in the description of the software used for the computations~\cite{lib}.

\paragraph{Organisation.} Section~\ref{sec:prelim} collects definitions and the ensemble form.
Section~\ref{sec:envelope} proves Theorem~B(i), Section~\ref{sec:n2} Theorem~A and Section~\ref{sec:bounds}
Theorem~B(ii). Section~\ref{sec:numerics} describes the computations for $n\ge3$ and Section~\ref{sec:open} lists open problems.

\section{Preliminaries}\label{sec:prelim}

\subsection{Conditional expectations and the entropy}

Throughout, $\M=M_n$, $\tau=\Tr/n$, and $\langle x,y\rangle_\tau=\tau(x^*y)$. Subalgebras are unital $*$-subalgebras. For such
a subalgebra $P$ the $\tau$-preserving conditional expectation $E_P$ is the orthogonal projection of $(\M,\langle\cdot,\cdot
\rangle_\tau)$ onto $P$; it is unital, positive, $P$-bimodular, preserves $\tau$ and $\Tr$, and is the unique conditional
expectation onto $P$ with these properties (see e.g.~\cite{takesaki2002}). The function $\eta(t)=-t\log t$, $\eta(0)=0$, is
continuous and concave on $[0,\infty)$; for $x\ge0$, $\eta(x)$ is defined by functional calculus, and
$S(\sigma)=\Tr\eta(\sigma)$ is the von Neumann entropy of a density matrix $\sigma$. The state space
$\D=\{\rho\in M_n:\rho\ge0,\ \Tr\rho=1\}$ is a compact convex set of real dimension $n^2-1$.

\begin{definition}\label{def:cs}
For subalgebras $P,Q\subset\M$ and $x\ge0$ put $F(x)=F_{P,Q}(x)=\tau\eta(E_Qx)-\tau\eta(E_Px)$. The Connes--St{\o}rmer relative
entropy is $H(P|Q)=\sup\sum_jF(x_j)$ over all finite families $x_1,\dots,x_m\ge0$ in $\M$ with $\sum_jx_j=1$ (partitions
of unity).
\end{definition}

\begin{lemma}\label{lem:F}
Let $P,Q\subset\M$ be subalgebras.
\begin{enumerate}
\item[(a)] $F(\lambda x)=\lambda F(x)$ for $x\ge0$ and $\lambda\ge0$. In particular, merging two elements of a partition
that are multiples of the same positive operator does not change $\sum_jF(x_j)$.
\item[(b)] Let $v\in\M$ be unitary and $\gamma=\Ad v$. Then $E_{\gamma(B)}=\gamma\circ E_B\circ\gamma^{-1}$ for every subalgebra
$B$, $F_{\gamma(P),\gamma(Q)}(\gamma(x))=F_{P,Q}(x)$, and $H(\gamma(P)|\gamma(Q))=H(P|Q)$; $\gamma$ maps optimal partitions for
$(P,Q)$ onto optimal partitions for $(\gamma(P),\gamma(Q))$.
\end{enumerate}
\end{lemma}
\begin{proof}
(a) For $s,\lambda\ge0$, $\eta(\lambda s)=\lambda\eta(s)+s\,\eta(\lambda)$, hence $\tau\eta(\lambda y)=\lambda\tau\eta(y)+\eta(\lambda)\tau(y)$
for $y\ge0$. Since $\tau(E_Qx)=\tau(E_Px)=\tau(x)$, the terms with $\eta(\lambda)$ cancel in $F(\lambda x)$. If
$x_1=a\,y$ and $x_2=b\,y$ with $a,b\ge0$, then $F(x_1)+F(x_2)=(a+b)F(y)=F(x_1+x_2)$.

(b) $\gamma$ is unitary for $\langle\cdot,\cdot\rangle_\tau$, since $\tau(\gamma(x)^*\gamma(y))=\tau(vx^*yv^*)=\tau(x^*y)$, and
$\gamma(B)$ is the image of $B$; so $\gamma E_B\gamma^{-1}$ is the orthogonal projection onto $\gamma(B)$. Moreover
$\eta(\gamma(y))=\gamma(\eta(y))$ and $\tau\circ\gamma=\tau$, so $\tau\eta(E_{\gamma(B)}\gamma(x))=\tau\gamma(\eta(E_Bx))=\tau\eta(E_Bx)$.
Since $\gamma$ maps partitions of unity bijectively onto partitions of unity, the remaining claims follow.
\end{proof}

\subsection{Two masas}

Let $\Delta_n$ be the diagonal masa and $Q_U=U\Delta_nU^*$ for a unitary $U$ with columns $u_1,\dots,u_n$. The minimal
projections of $Q_U$ are $p_k=u_ku_k^*$, and
\begin{equation}\label{eq:EQ}
  E_{\Delta_n}(x)=\sum_ke_{kk}xe_{kk},\qquad E_{Q_U}(x)=\sum_kp_kxp_k=\sum_k\langle u_k,xu_k\rangle\,p_k,
\end{equation}
since $\sum_kp_kxp_k\in Q_U$ and $\tau\bigl(p_l(x-\sum_kp_kxp_k)\bigr)=\tau(p_lx)-\tau(p_lxp_l)=0$ for every $l$. Thus the
spectrum of $E_{Q_U}x$ is the diagonal of $U^*xU$, and
\begin{equation}\label{eq:Fmasa}
  F(x)=\frac1n\Bigl[\sum_k\eta\bigl((U^*xU)_{kk}\bigr)-\sum_k\eta(x_{kk})\Bigr].
\end{equation}
The algebra $Q_U$ does not change if $U$ is multiplied from the right by a diagonal unitary or a permutation matrix, and by
Lemma~\ref{lem:F}(b) with $\gamma=\Ad D$ for a diagonal unitary $D$ (which fixes $\Delta_n$), $H(\Delta_n|Q_{DU})=H(\Delta_n|Q_U)$.
For the partition $(e_{kk})_k$, $\eta(x_{kk})\in\{\eta(0),\eta(1)\}=\{0\}$ and $(U^*e_{kk}U)_{jj}=|u_{kj}|^2$, which
gives~\eqref{eq:choda}. For an orthonormal basis $w_1,\dots,w_n$, the columns of a unitary $W$, the partition
$x_j=w_jw_j^*$ gives, by~\eqref{eq:Fmasa},
\begin{equation}\label{eq:basis}
  \sum_jF(w_jw_j^*)=h(U^*W)-h(W),
\end{equation}
and Choda's partition is the case $W=1$.

\subsection{The ensemble form}

The following reformulation is due to Petz, Sz\'ant\'o and Weiner~\cite[Section~3, Lemma~1 and (7), (8)]{petz2009}; we
include the short proof, and a Carath\'eodory argument for the attainment of the supremum.

\begin{proposition}[Petz--Sz\'ant\'o--Weiner]\label{prop:psw}
Let $P,Q\subset M_n$ be subalgebras and $G(\rho)=S(E_Q\rho)-S(E_P\rho)$ for $\rho\in\D$.
\begin{enumerate}
\item[(a)] A partition $(x_j)$ with $x_j\ne0$ corresponds to the ensemble $t_j=\tau(x_j)$, $\rho_j=x_j/\Tr x_j$, which satisfies
$t_j>0$, $\sum_jt_j=1$ and $\sum_jt_j\rho_j=\frac1n$; conversely every such ensemble defines the partition $x_j=nt_j\rho_j$.
In both directions $\sum_jF(x_j)=\sum_jt_jG(\rho_j)$.
\item[(b)] $H(P|Q)=\max\bigl\{\sum_jt_jG(\rho_j):\ t_j\ge0,\ \sum_jt_j=1,\ \sum_jt_j\rho_j=\frac1n\bigr\}$; the maximum is attained by
an ensemble of at most $n^2+1$ states. (Petz, Sz\'ant\'o and Weiner show that $n^2$ linearly independent states suffice.)
\item[(c)] $G\le\log n$ on $\D$, hence $H(P|Q)\le\log n$.
\end{enumerate}
\end{proposition}
\begin{proof}
(a) Elements $x_j=0$ contribute $F(0)=0$ and can be dropped. $\Tr x_j=n\tau(x_j)=nt_j$, so $x_j=nt_j\rho_j$, and $\sum_jx_j=1$ is equivalent to $\sum_jt_j\rho_j=\frac1n$, which implies
$\sum_jt_j=1$ by taking traces. Since $E_P\rho$ and $E_Q\rho$ are density matrices, $F(\rho)=\frac1n[S(E_Q\rho)-S(E_P\rho)]=\frac1nG(\rho)$,
and Lemma~\ref{lem:F}(a) gives $F(x_j)=nt_jF(\rho_j)=t_jG(\rho_j)$.

(b) By (a), $H$ is the supremum of $\sum_jt_jG(\rho_j)$ over finite ensembles with barycentre $\frac1n$. Identify the affine
hull of $\D$ with $\R^{n^2-1}$. For such an ensemble the point $(\frac1n,\sum_jt_jG(\rho_j))\in\R^{n^2}$ is a convex combination of
points of the graph $\{(\rho,G(\rho)):\rho\in\D\}$; by Carath\'eodory's theorem it is a convex combination of at most $n^2+1$
of them, i.e.\ there is an ensemble of at most $n^2+1$ states with the same barycentre and the same value. The ensembles of
$n^2+1$ states (zero weights allowed) with barycentre $\frac1n$ form a compact set, on which $\sum_jt_jG(\rho_j)$ is continuous,
because $S$ is continuous on $\D$ and $E_P,E_Q$ are linear. So the supremum is a maximum.

(c) $S(E_Q\rho)\le\log n$ for the density matrix $E_Q\rho$ of $M_n$, and $S(E_P\rho)\ge0$.
\end{proof}

For two masas, $G(\rho)=H_{\rm Sh}\bigl(\mathrm{diag}(U^*\rho U)\bigr)-H_{\rm Sh}\bigl(\mathrm{diag}\,\rho\bigr)$ with the Shannon
entropy $H_{\rm Sh}$ of a probability vector, by~\eqref{eq:EQ}.

\section{The concave envelope and the bound \texorpdfstring{$G^*$}{G*}}\label{sec:envelope}

For a bounded function $g$ on $\D$ let $\hat g(\rho)=\sup\{\sum_jt_jg(\rho_j):\ \text{finite ensembles with }\sum_jt_j\rho_j=\rho\}$;
this is the smallest concave function on $\D$ that majorises $g$ (its concave envelope).

\begin{proposition}\label{prop:envelope}
Let $P,Q\subset M_n$ be subalgebras, $G$ as in Proposition~\ref{prop:psw}, $G^*=\max_\D G$ and $A=\{\rho\in\D:G(\rho)=G^*\}$.
Then
\[
  H(P|Q)=\hat G\bigl(\tfrac1n\bigr)\le G^*,
\]
and $H(P|Q)=G^*$ if and only if $\frac1n\in\conv A$. In that case the optimal partitions are exactly the partitions
$x_j=nt_j\rho_j$ with $\rho_j\in A$ whenever $t_j>0$.
\end{proposition}
\begin{proof}
The identity $H=\hat G(\frac1n)$ is Proposition~\ref{prop:psw}(b) with the definition of $\hat G$. For every ensemble,
$\sum_jt_jG(\rho_j)\le\sum_jt_jG^*=G^*$, with equality if and only if $G(\rho_j)=G^*$ whenever $t_j>0$. If $\frac1n=\sum_jt_j\rho_j$
with $\rho_j\in A$, this ensemble has the value $G^*$, so $H=G^*$. Conversely, if $H=G^*$, the maximum in
Proposition~\ref{prop:psw}(b) is attained by an ensemble with value $G^*$, all of whose states with positive weight lie in
$A$; hence $\frac1n\in\conv A$. The last statement is the equality case together with Proposition~\ref{prop:psw}(a).
\end{proof}

\begin{remark}\label{rem:envelope}
(i) $G^*$ is a maximisation over the $(n^2-1)$-dimensional state space, whereas $H$ requires ensembles of up to $n^2$ states
with a constraint; numerically $G^*$ is much cheaper. Any state gives a lower bound for $G^*$, any partition a lower bound
for $H$.
(ii) For two masas of $M_2$, $H=G^*$ (Theorem~\ref{thm:n2}). For $n=3,4,5$ the computations of Section~\ref{sec:numerics}
find $G^*>H$ for every sampled unitary, i.e.\ the optimal ensembles do not consist of maximisers of $G$.
(iii) If $U$ is complementary, $G(e_{11})=\log n=G^*$ by~\eqref{eq:Fmasa}, and $\frac1n=\frac1n\sum_ke_{kk}$ is a convex
combination of the maximisers $e_{kk}$ (all of which satisfy $G(e_{kk})=\log n$); so $H=G^*=\log n$, the partition
$(e_{kk})$ is optimal, and $h(U)=\log n$.
\end{remark}

\section{Two masas of \texorpdfstring{$M_2$}{M2}}\label{sec:n2}

\subsection{The function \texorpdfstring{$\psi$}{psi}}

Let $\Psi(\beta)=h(\cos^2\beta)$ and let $\psi$ be as in~\eqref{eq:psi}.

\begin{lemma}\label{lem:psi}
\begin{enumerate}
\item[(a)] $\psi$ is continuous on $\R$, odd, of period $\frac\pi2$, vanishes exactly on $\frac\pi4\Z$, and $\psi>0$ on
$(-\frac\pi4,0)$, $\psi<0$ on $(0,\frac\pi4)$.
\item[(b)] $\Psi$ is continuously differentiable on $\R$, $\Psi'=-\psi$, and $\Psi$ is even and of period $\frac\pi2$.
\item[(c)] On $(-\frac\pi4,0)$, $\psi$ has a unique critical point $\beta^*$; $\psi$ is strictly increasing on $[-\frac\pi4,\beta^*]$ and
strictly decreasing on $[\beta^*,0]$. With $\beta^*=-s^*/2$ and $y=\cos s^*$, $y\in(0,1)$ is the unique solution of
$y\artanh y=1$, and
\[
  \kappa:=\max_\R\psi=\psi(\beta^*)=2\tan s^*=\frac{2\sqrt{1-y^2}}{y},\qquad \psi''(\beta^*)=-4\kappa-\frac{16}\kappa .
\]
\end{enumerate}
\end{lemma}
\begin{proof}
(a) For $\beta\notin\frac\pi2\Z$, $\psi(-\beta)=-\psi(\beta)$ since $\sin$ is odd and $\tan^2$ even, and
$\psi(\beta+\frac\pi2)=\psi(\beta)$ since $\sin(2\beta+\pi)=-\sin2\beta$ and $\tan^2(\beta+\frac\pi2)=\cot^2\beta$, so that
$\log\tan^2(\beta+\frac\pi2)=-\log\tan^2\beta$. Near $0$, $|\psi(\beta)|\le2|\beta|\,|\log\tan^2\beta|\to0$, so $\psi$ is
continuous at $0$ with $\psi(0)=0$, and by periodicity at every point of $\frac\pi2\Z$. On $(0,\frac\pi4)$, $\sin2\beta>0$ and
$0<\tan^2\beta<1$, so $\psi<0$; by oddness $\psi>0$ on $(-\frac\pi4,0)$. $\psi(\pm\frac\pi4)=\pm\log1=0$. By periodicity these
are all zeros.

(b) $h$ is smooth on $(0,1)$ with $h'(p)=\log\frac{1-p}p$. For $\beta\notin\frac\pi2\Z$, $0<\cos^2\beta<1$ and
$\Psi'(\beta)=h'(\cos^2\beta)\cdot(-\sin2\beta)=-\sin2\beta\log\tan^2\beta=-\psi(\beta)$. At $\beta_0\in\frac\pi2\Z$, $\Psi$ is
continuous, and by the mean value theorem $(\Psi(\beta)-\Psi(\beta_0))/(\beta-\beta_0)=-\psi(\xi)$ for some $\xi$ between
$\beta_0$ and $\beta$; by continuity of $\psi$ this tends to $-\psi(\beta_0)=0$. Hence $\Psi\in C^1(\R)$, $\Psi'=-\psi$. Evenness and
periodicity follow from $\cos^2(-\beta)=\cos^2\beta$ and $h(\cos^2(\beta+\frac\pi2))=h(\sin^2\beta)=h(\cos^2\beta)$.

(c) Away from $\frac\pi2\Z$, $(\log\tan^2\beta)'=2/(\sin\beta\cos\beta)=4/\sin2\beta$, hence
\[
  \psi'(\beta)=2\cos2\beta\log\tan^2\beta+4,\qquad \psi''(\beta)=-4\psi(\beta)+8\cot2\beta .
\]
For $b\in(0,\frac\pi4)$ let $g(b)=\cos2b\log\tan b$, so that $\psi'(-b)=4(g(b)+1)$. Then $g'(b)=-2\sin2b\log\tan b+2\cos2b/\sin2b>0$,
$g(b)\to-\infty$ as $b\to0$ and $g(\frac\pi4)=0$. Hence $g=-1$ at exactly one point $b^*$, $\psi'>0$ on $(-\frac\pi4,-b^*)$ and
$\psi'<0$ on $(-b^*,0)$; put $\beta^*=-b^*$, $s^*=2b^*\in(0,\frac\pi2)$. Since $\log\tan b=-\artanh(\cos2b)$ (both sides equal
$\frac12\log\frac{1-\cos2b}{1+\cos2b}$), the equation $g(b^*)=-1$ reads $y\artanh y=1$ with $y=\cos s^*$, and $y\artanh y$ increases
strictly from $0$ to $\infty$ on $(0,1)$. Then $\psi(\beta^*)=-\sin s^*\cdot2\log\tan b^*=2\sin s^*/\cos s^*=2\tan s^*$. By (a)
the maximum of $\psi$ over a period is attained in $[-\frac\pi4,0]$, hence at $\beta^*$. Finally
$\psi''(\beta^*)=-4\kappa+8\cot(-s^*)=-4\kappa-8/\tan s^*=-4\kappa-16/\kappa$.
\end{proof}

\noindent Numerically $\kappa=\CSKappa$ (computed by \texttt{scripts/connes\_\allowbreak stormer.py}; the test suite checks
$\kappa=1.3254868$ to $10^{-6}$).

\subsection{The closed form}

Write $R_\theta=\bigl(\begin{smallmatrix}\cos\theta&-\sin\theta\\ \sin\theta&\cos\theta\end{smallmatrix}\bigr)$.

\begin{lemma}\label{lem:normal}
Every unitary $U\in M_2$ can be written as $U=D_1R_\theta D_2$ with diagonal unitaries $D_1,D_2$ and $\theta\in[0,\frac\pi2]$,
$\cos^2\theta=|u_{11}|^2$. Consequently $H(\Delta_2|U\Delta_2U^*)=H(\Delta_2|R_\theta\Delta_2R_\theta^T)$, and $\gamma=\Ad D_1$ maps
optimal partitions for $R_\theta$ onto optimal partitions for $U$.
\end{lemma}
\begin{proof}
The columns and the rows of $U$ are unit vectors, so $|u_{11}|^2+|u_{21}|^2=1=|u_{11}|^2+|u_{12}|^2$ and
$|u_{12}|^2+|u_{22}|^2=1$; hence $|u_{21}|=|u_{12}|$ and $|u_{22}|=|u_{11}|$. Let $\theta\in[0,\frac\pi2]$ with $\cos\theta=|u_{11}|$
and write $u_{11}=e^{ia}\cos\theta$, $u_{21}=e^{ib}\sin\theta$, $u_{12}=-e^{ic}\sin\theta$, $u_{22}=e^{id}\cos\theta$. If
$0<\theta<\frac\pi2$, orthogonality of the columns gives $(e^{i(d-b)}-e^{i(c-a)})\cos\theta\sin\theta=0$, i.e.\
$e^{i(d-b)}=e^{i(c-a)}$, and then $U=\mathrm{diag}(e^{ia},e^{ib})\,R_\theta\,\mathrm{diag}(1,e^{i(c-a)})$, as one checks entry by
entry. If $\theta=0$, $U$ is diagonal and $U=U\cdot R_0\cdot1$; if $\theta=\frac\pi2$, $U=\mathrm{diag}(-u_{12},u_{21})R_{\pi/2}$.
Since $D_2\Delta_2D_2^*=\Delta_2$, $U\Delta_2U^*=\gamma(R_\theta\Delta_2R_\theta^T)$ with $\gamma=\Ad D_1$, which fixes
$\Delta_2$; Lemma~\ref{lem:F}(b) gives the rest.
\end{proof}

We use the Pauli matrices $\sigma=(\sigma_1,\sigma_2,\sigma_3)$ and the Bloch parametrisation
$\rho=\frac12(1+r\cdot\sigma)$, $r\in\R^3$, $|r|\le1$, of $\D$ for $n=2$; $\rho$ is a rank-one projection if and only if $|r|=1$. For a
unit vector $v=(\cos a,\sin a)^T$ one has $vv^*=\frac12(1+\sin2a\,\sigma_1+\cos2a\,\sigma_3)$, so its Bloch vector is
$(\sin2a,0,\cos2a)$, and the Bloch vector of $(-\sin a,\cos a)^T$ is the negative of it. Let
\[
  z=(0,0,1),\qquad q=(\sin2\theta,0,\cos2\theta),\qquad k(s)=h\bigl(\tfrac{1+s}2\bigr)\ (|s|\le1),
\]
so $z$ and $q$ are the Bloch vectors of $e_1$ and of $R_\theta e_1$, and $k$ is even.

\begin{lemma}\label{lem:bloch}
For $P=\Delta_2$, $Q=R_\theta\Delta_2R_\theta^T$ and $\rho=\frac12(1+r\cdot\sigma)$, $G(\rho)=f(r):=k(r\cdot q)-k(r\cdot z)$. Moreover
\begin{equation}\label{eq:series}
  k(s)=\log2-\sum_{m\ge1}\frac{s^{2m}}{2m(2m-1)}\qquad(|s|\le1),
\end{equation}
the series converging uniformly on $[-1,1]$.
\end{lemma}
\begin{proof}
$E_{\Delta_2}\rho=\mathrm{diag}(\rho)$ has the eigenvalues $\frac12(1\pm r_3)=\frac12(1\pm r\cdot z)$, so $S(E_{\Delta_2}\rho)=k(r\cdot z)$. By~\eqref{eq:EQ}
the eigenvalues of $E_Q\rho$ are $\langle v_1,\rho v_1\rangle$ and $\langle v_2,\rho v_2\rangle=1-\langle v_1,\rho v_1\rangle$ with
$v_1=R_\theta e_1$, and $\langle v_1,\rho v_1\rangle=\Tr(\rho v_1v_1^*)=\frac12(1+r\cdot q)$ by $\Tr\bigl((1+r\cdot\sigma)(1+q\cdot\sigma)\bigr)
=2(1+r\cdot q)$. So $S(E_Q\rho)=k(r\cdot q)$.

For~\eqref{eq:series}, $k(s)=\log2-\frac12\bigl[(1+s)\log(1+s)+(1-s)\log(1-s)\bigr]$. For $|s|<1$, integrating
$\frac{d^2}{ds^2}(1\pm s)\log(1\pm s)=1/(1\pm s)=\sum_{j\ge0}(\mp s)^j$ twice gives $(1\pm s)\log(1\pm s)=\pm s+\sum_{l\ge2}(\mp1)^ls^l/(l(l-1))$,
and adding the two expansions leaves $2\sum_{m\ge1}s^{2m}/(2m(2m-1))$. The terms are bounded by $1/(2m(2m-1))$ on $[-1,1]$, so
the series converges uniformly there (Weierstrass), its sum is continuous, and the identity extends to $s=\pm1$ by
continuity of $k$.
\end{proof}

\begin{theorem}\label{thm:n2}
Let $U\in M_2$ be unitary, $U=D_1R_\theta D_2$ as in Lemma~\ref{lem:normal}, $c=|u_{11}|^2=\cos^2\theta$, and
\[
  \varphi(\alpha)=h\bigl(\cos^2(\alpha-\theta)\bigr)-h(\cos^2\alpha)=\int_{\alpha-\theta}^{\alpha}\psi(\beta)\,d\beta .
\]
Then $\varphi$ is $C^1$ and $\frac\pi2$-periodic, $\varphi'(\alpha)=\psi(\alpha)-\psi(\alpha-\theta)$, and
\begin{enumerate}
\item[(a)] $H(\Delta_2|U\Delta_2U^*)=\max_{\alpha\in\R}\varphi(\alpha)=G^*$;
\item[(b)] for every maximiser $\alpha^*$ of $\varphi$, the projections onto the orthonormal basis $D_1(\cos\alpha^*,\sin\alpha^*)^T$,
$D_1(-\sin\alpha^*,\cos\alpha^*)^T$ form an optimal partition;
\item[(c)] $H(\Delta_2|U\Delta_2U^*)$ depends only on $c$, and it is invariant under $c\mapsto1-c$.
\end{enumerate}
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:normal} we may assume $U=R_\theta$, $D_1=1$, and transport optimal partitions by $\Ad D_1$ at the end.
The integral form is $\varphi(\alpha)=\Psi(\alpha-\theta)-\Psi(\alpha)=\int_{\alpha-\theta}^\alpha\psi$ by Lemma~\ref{lem:psi}(b), which
also gives the regularity, periodicity and derivative of $\varphi$.

\emph{Step 1: $H=G^*=\max_{|r|\le1}f(r)$.} By Lemma~\ref{lem:bloch}, $G^*=\max_{|r|\le1}f$, the maximum of a continuous function on
the closed unit ball. Since $k$ is even, $f(-r)=f(r)$; so if $r^*$ maximises $f$, the states $\rho_\pm=\frac12(1\pm r^*\cdot\sigma)$
both maximise $G$, and $\frac12\rho_++\frac12\rho_-=\frac12\cdot1$. By Proposition~\ref{prop:envelope}, $H=G^*$, and the partition
$x_\pm=2\cdot\frac12\rho_\pm=\rho_\pm$ is optimal.

\emph{Step 2: reduction to the unit circle in the plane of $z$ and $q$.} If $\theta\in\{0,\frac\pi2\}$, then $q=\pm z$ and $f\equiv0$
since $k$ is even; also $\varphi\equiv0$, because $h(\cos^2(\alpha-\frac\pi2))=h(\sin^2\alpha)=h(\cos^2\alpha)$. So $H=0=\max\varphi$
and every basis is optimal. Let $0<\theta<\frac\pi2$. Then $z$ and $q$ span the plane $\{r_2=0\}$, and $f(r)$ depends only on
the orthogonal projection $a=(r_1,0,r_3)$ of $r$ onto it, $|a|\le|r|\le1$. Write $a=\varrho(\sin\beta,0,\cos\beta)$ with
$0\le\varrho\le1$; then $a\cdot z=\varrho\cos\beta$ and $a\cdot q=\varrho\cos(\beta-2\theta)$, and by~\eqref{eq:series}
\[
  f(a)=\sum_{m\ge1}\frac{\varrho^{2m}}{2m(2m-1)}\bigl(\cos^{2m}\beta-\cos^{2m}(\beta-2\theta)\bigr).
\]
If $|\cos\beta|\ge|\cos(\beta-2\theta)|$, all terms are non-negative and non-decreasing in $\varrho$, so
$f(a)\le f(\sin\beta,0,\cos\beta)$. Otherwise all terms are non-positive and $f(a)\le0=f(\sin\theta,0,\cos\theta)$, the
value at $\beta=\theta$, where $\cos\beta=\cos(\beta-2\theta)$. Hence the maximum of $f$ over the ball equals its maximum over the
unit circle of the plane,
\[
  \max_{|r|\le1}f=\max_{\beta\in\R}\bigl[k(\cos(\beta-2\theta))-k(\cos\beta)\bigr],
\]
and it is attained at a unit vector $r^*=(\sin\beta^\circ,0,\cos\beta^\circ)$.

\emph{Step 3: the angle.} Since $k(\cos2\gamma)=h(\frac{1+\cos2\gamma}2)=h(\cos^2\gamma)$, the substitution $\beta=2\alpha$ turns
the right-hand side into $\max_\alpha\varphi(\alpha)$. This proves (a). For a maximiser $\alpha^*$ of $\varphi$, $r^*=(\sin2\alpha^*,0,
\cos2\alpha^*)$ maximises $f$, and $\rho_+$, $\rho_-$ are the projections onto $(\cos\alpha^*,\sin\alpha^*)^T$ and
$(-\sin\alpha^*,\cos\alpha^*)^T$; with Step~1 and Lemma~\ref{lem:normal} this proves (b).

(c) $\varphi$ depends only on $\theta$, i.e.\ on $c$. Let $J=\bigl(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\bigr)$. Then
$UJ\Delta_2(UJ)^*=U\Delta_2U^*$ and $|(UJ)_{11}|^2=|u_{12}|^2=1-c$.
\end{proof}

\begin{remark}\label{rem:n2}
(i) At $\alpha=0$, $\varphi(0)=h(c)$, the value of Choda's partition $e_{11},e_{22}$; indeed $h(U)=h(c)$ for $n=2$. At $\alpha=\theta$
(the basis of $Q$) $\varphi(\theta)=-h(c)$.
(ii) In the Bloch picture, $z$ and $q$ enclose the angle $2\theta$, and the optimal Bloch vector $r^*$ has the angle $2\alpha^*$ from
$z$. By Proposition~\ref{prop:angle} below, for $\frac12<c<1$ it lies on the other side of $z$ than $q$: the optimal partition
accepts some entropy of $E_{\Delta_2}x_j$ in exchange for more entropy of $E_Qx_j$.
(iii) Let $0<c<1$. Since $H=G^*$, the optimal partitions are exactly the families $x_j=2t_j\rho_j$ with $\rho_j$ maximising
$G$ and $\sum_jt_j\rho_j=\frac12$ (Proposition~\ref{prop:envelope}). Every maximiser $r$ of $f$ has $|r|=1$ and $r_2=0$: as
$f(r)=H\ge h(c)>0$, in Step~2 of the proof we are in the case $|\cos\beta|>|\cos(\beta-2\theta)|$, where every term of the
series is strictly increasing in $\varrho$, and $r_2\ne0$ or $|r|<1$ would give $\varrho<1$. Since the maximiser of $\varphi$ is
unique modulo $\frac\pi2$ (Proposition~\ref{prop:angle}), the maximisers of $G$ are the two pure states $\rho_\pm$, and the
optimal partition is unique up to splitting $\rho_\pm$ into several multiples (Lemma~\ref{lem:F}(a)).
\end{remark}

\subsection{The optimal angle}

\begin{proposition}\label{prop:angle}
Let $\frac12<c<1$, i.e.\ $0<\theta<\frac\pi4$, and let $\beta^*$ be as in Lemma~\ref{lem:psi}(c). Then $\varphi$ has a unique
maximiser $\alpha^*$ in $(-\frac\pi4,\frac\pi4]$; it satisfies
\[
  \max\bigl(\beta^*,\theta-\tfrac\pi4\bigr)<\alpha^*<\min\bigl(\beta^*+\theta,0\bigr),
\]
and it is the unique solution of $\psi(\alpha)=\psi(\alpha-\theta)$ in this interval. In particular $\alpha^*<0$, and
$\alpha^*\to\beta^*$ as $\theta\to0$. For $0<c<\frac12$ the maximiser is $\alpha^*(c)=-\alpha^*(1-c)$. For $c=\frac12$ the
maximiser is $\alpha^*=0$ (modulo $\frac\pi2$) and $H=\log2$.
\end{proposition}
\begin{proof}
Let $I_k=(-\frac\pi4+k\frac\pi2,k\frac\pi2)$ and $N_k=(k\frac\pi2,\frac\pi4+k\frac\pi2)$, $k\in\Z$; by Lemma~\ref{lem:psi}(a), $\psi>0$
on the $I_k$ and $\psi<0$ on the $N_k$, all of length $\frac\pi4$. Let $0<\theta\le\frac\pi4$ and let $W=[\alpha-\theta,\alpha]$ be a window. If
$W$ met two of the $I_k$, it would contain the closure of some $N_k$ together with points on both sides of it, so its length
would exceed $\frac\pi4$; hence $W$ meets at most one $I_k$, in an interval $J$ of length at most $\theta$. Choose a window $W'\subset\overline{I_k}$ of length $\theta$
containing $J$. Since $\psi\ge0$ on $\overline{I_k}$ and $\psi\le0$ on $W\setminus J$,
\[
  \int_W\psi\le\int_J\psi\le\int_{W'}\psi,
\]
and the first inequality is strict unless $W\setminus J$ is a null set, i.e.\ unless $W\subset\overline{I_k}$. Hence every
maximising window lies in some $\overline{I_k}$; modulo $\frac\pi2$, $W\subset[-\frac\pi4,0]$, i.e.\ $\alpha\in[\theta-\frac\pi4,0]$.

\emph{$c=\frac12$.} For $\theta=\frac\pi4$ the only such window is $[-\frac\pi4,0]$, so $\alpha^*=0$ and
$H=\Psi(-\frac\pi4)-\Psi(0)=h(\frac12)-h(1)=\log2$.

\emph{$\frac12<c<1$.} Now $0<\theta<\frac\pi4$. At the end points, $\varphi'(0)=\psi(0)-\psi(-\theta)=-\psi(-\theta)<0$ and
$\varphi'(\theta-\frac\pi4)=\psi(\theta-\frac\pi4)-\psi(-\frac\pi4)=\psi(\theta-\frac\pi4)>0$, so neither is a maximiser of the
$C^1$ function $\varphi$ on $\R$, and every maximiser $\alpha^*$ (modulo $\frac\pi2$) lies in $(\theta-\frac\pi4,0)$ and satisfies
$\varphi'(\alpha^*)=0$, i.e.\ $\psi(\alpha^*)=\psi(\alpha^*-\theta)$ with $\alpha^*-\theta<\alpha^*$ both in $(-\frac\pi4,0)$. As $\psi$ is
strictly monotone on $[-\frac\pi4,\beta^*]$ and on $[\beta^*,0]$ (Lemma~\ref{lem:psi}(c)), two distinct points with equal values lie
on different sides of $\beta^*$: $\alpha^*-\theta<\beta^*<\alpha^*$. On the interval $L=(\max(\beta^*,\theta-\frac\pi4),
\min(\beta^*+\theta,0))$ we have $\alpha\in(\beta^*,0)$ and $\alpha-\theta\in(-\frac\pi4,\beta^*)$, so $\psi(\alpha)$ is strictly
decreasing and $\psi(\alpha-\theta)$ strictly increasing in $\alpha$; hence $\varphi'$ is strictly decreasing on $L$ and has at most
one zero there. A maximiser exists by continuity and periodicity, so it is unique modulo $\frac\pi2$. As $\theta\to0$,
$L\subset(\beta^*,\beta^*+\theta)$ shrinks to $\beta^*$.

\emph{$0<c<\frac12$.} With $\theta'=\frac\pi2-\theta$ and Lemma~\ref{lem:psi}(b),
$\varphi_\theta(\alpha)=\Psi(\alpha-\frac\pi2+\theta')-\Psi(\alpha)=\Psi(-\alpha-\theta')-\Psi(-\alpha)=\varphi_{\theta'}(-\alpha)$,
where $\varphi_\theta$ denotes $\varphi$ for the angle $\theta$; and $\cos^2\theta'=1-c$.
\end{proof}

\noindent At $\theta\to0$ the optimal basis therefore does not approach the basis of $\Delta_2$ (angle $0$) or of $Q$ (angle
$\theta$), but the fixed angle $\beta^*$; the middle panel of Figure~\ref{fig:cs} shows $\alpha^*$ against $c$.

\section{Bounds for \texorpdfstring{$n=2$}{n=2}}\label{sec:bounds}

\begin{theorem}\label{thm:bound}
Let $U\in M_2$ be unitary, $c=|u_{11}|^2=\cos^2\theta$, $\theta\in[0,\frac\pi2]$, and $\kappa=\max\psi$ as in
Lemma~\ref{lem:psi}(c). Then
\begin{enumerate}
\item[(a)] $h(c)\le H(\Delta_2|U\Delta_2U^*)\le\kappa\min\bigl(\theta,\frac\pi2-\theta\bigr)$;
\item[(b)] as $\theta\to0$,
$H(\Delta_2|U\Delta_2U^*)=\kappa\theta-\frac{\kappa+4/\kappa}6\theta^3+O(\theta^4)$, while $h(c)=o(\theta)$; in particular
$H/h(U)\to\infty$ as $c\to1$, and likewise as $c\to0$.
\end{enumerate}
\end{theorem}
\begin{proof}
(a) The lower bound is $\varphi(0)=h(c)$. For every $\alpha$, $\varphi(\alpha)=\int_{\alpha-\theta}^\alpha\psi\le\theta\max\psi=\kappa\theta$, so
$H\le\kappa\theta$ by Theorem~\ref{thm:n2}(a). Applying this to $1-c$, whose angle is $\frac\pi2-\theta$, and using
Theorem~\ref{thm:n2}(c) gives $H\le\kappa(\frac\pi2-\theta)$.

(b) Fix $\delta>0$ with $[\beta^*-\delta,\beta^*+\delta]\subset(-\frac\pi4,0)$; $\psi$ is smooth there, and by Taylor's theorem
there is $C$ with $|\psi(\beta)-\kappa-\frac12\psi''(\beta^*)(\beta-\beta^*)^2|\le C|\beta-\beta^*|^3$ for $|\beta-\beta^*|\le\delta$. Let
$\theta<\min(\delta,\frac\pi4)$. By Proposition~\ref{prop:angle} the maximising window $W=[\alpha^*-\theta,\alpha^*]$ contains
$\beta^*$, so $W\subset[\beta^*-\delta,\beta^*+\delta]$. Since $\int_W(\beta-\beta^*)^2d\beta\ge\theta^3/12$ for every window of length
$\theta$ (with equality for the window centred at $\beta^*$), $\psi''(\beta^*)<0$ and $\int_W|\beta-\beta^*|^3d\beta\le\theta^4$,
\[
  \kappa\theta+\frac{\psi''(\beta^*)}{24}\theta^3-C\theta^4\ \le\ \varphi\bigl(\beta^*+\tfrac\theta2\bigr)\ \le\ H\ =\ \int_W\psi\ \le\
  \kappa\theta+\frac{\psi''(\beta^*)}{24}\theta^3+C\theta^4 ,
\]
and $\psi''(\beta^*)/24=-(\kappa+4/\kappa)/6$ by Lemma~\ref{lem:psi}(c). Finally $h(c)=\varphi(0)=\int_{-\theta}^0\psi$ and
$|\int_{-\theta}^0\psi|\le\theta\max_{[-\theta,0]}|\psi|=o(\theta)$, as $\psi$ is continuous with $\psi(0)=0$. Since $h(U)=h(c)>0$
for $0<c<1$, $H/h(U)\ge(\kappa\theta+O(\theta^3))/o(\theta)\to\infty$. The case $c\to0$ follows from Theorem~\ref{thm:n2}(c).
\end{proof}

\noindent Since $\theta=\arccos\sqrt c\sim\sqrt{1-c}$, near $c=1$ the entropy behaves like $\kappa\sqrt{1-c}$, much larger than
$h(c)\sim(1-c)\log\frac1{1-c}$. The bound $\log2$ of Proposition~\ref{prop:psw}(c) is better than $\kappa\min(\theta,\frac\pi2-\theta)$
near $c=\frac12$; Table~\ref{tab:cs2} lists $\kappa\theta$ for comparison.

\begin{corollary}[Petz--Sz\'ant\'o--Weiner]\label{cor:choda}
$H(\Delta_2|U\Delta_2U^*)>h(U)$ unless $c\in\{0,\frac12,1\}$, where $H=h(U)$.
\end{corollary}
\begin{proof}
This is~\cite[Appendix]{petz2009}; in terms of Theorem~\ref{thm:n2} it reads as follows. $h(U)=h(c)=\varphi(0)$ and
$\varphi'(0)=\psi(0)-\psi(-\theta)=\psi(\theta)$, which by Lemma~\ref{lem:psi}(a) vanishes exactly for $\theta\in\{0,\frac\pi4,\frac\pi2\}$.
Otherwise $0$ is not a maximiser of $\varphi$, so $H>\varphi(0)$. For $c\in\{0,1\}$, $H=0=h(c)$, and for $c=\frac12$, $H=\log2=h(\frac12)$
(Proposition~\ref{prop:angle}).
\end{proof}

\begin{remark}[Spin model subfactors]\label{rem:bakshi}
Bakshi and Guin use the inequality $H(u\Delta_2u^*|v\Delta_2v^*)\le H(R_u|R_v)$ for the spin model subfactors $R_u,R_v$ of two
$2\times2$ complex Hadamard matrices $u,v$~\cite[proof of Theorem~6.22]{bakshi}, together with Choda's bound. By
Lemma~\ref{lem:F}(b) with $\gamma=\Ad u^*$, $H(u\Delta_2u^*|v\Delta_2v^*)=H(\Delta_2|U\Delta_2U^*)$ with $U=u^*v$, so
Theorem~\ref{thm:n2} with $c=|(u^*v)_{11}|^2$ raises their lower bound $h(c)$ for $H(R_u|R_v)$ to $\max_\alpha\varphi(\alpha)$,
e.g.\ from $0.3251$ to $0.4024$ at $c=0.9$ (Table~\ref{tab:cs2}).
\end{remark}

\section{Numerical study for \texorpdfstring{$n\ge3$}{n>=3}}\label{sec:numerics}

\subsection{Method}

All computations use our PyTorch~\cite{paszke2019} library \texttt{torch\_\allowbreak vn\_\allowbreak algebra}~\cite{lib};
the program is \texttt{scripts/\allowbreak connes\_\allowbreak stormer.py} of the repository~\cite{lib}, in double precision
(complex128) on a CPU.

\emph{The functional.} $\Delta_n$ is the image of the inclusion $\C^n\subset M_n$ with inclusion matrix $(1,\dots,1)$, and $Q_U$
the image of the same inclusion twisted by $U$ (\texttt{inclusions.Inclusion} with \texttt{unitary=U}); the conditional
expectations onto both are taken for the normalised trace $\tau=\Tr/n$, and $\tau\eta(E(x))$ is computed from the eigenvalues of
$E(x)$. The whole computation is batched and differentiable. The test suite checks this path against the plain NumPy
formula~\eqref{eq:Fmasa} to $10^{-12}$ for random partitions with $n=2,3,5$.

\emph{Search spaces.} (1) General partitions with $m=n^2$ elements,
\[
  x_j=S^{-1/2}Z_jZ_j^*S^{-1/2},\qquad S=\sum_{j=1}^mZ_jZ_j^*,
\]
with free complex $Z_j\in M_n$; every family with invertible $S$ is a partition of unity, and every partition of unity with
$m$ elements arises this way ($Z_j=x_j^{1/2}$, $S=1$). By the result of Petz, Sz\'ant\'o and Weiner quoted in
Proposition~\ref{prop:psw}(b), $m=n^2$ elements suffice. (2) Orthonormal bases, $x_j=w_jw_j^*$ for
the columns $w_j$ of a unitary $W=W_0\exp(X-X^*)$, with value~\eqref{eq:basis}. (3) States $\rho=ZZ^*/\Tr(ZZ^*)$ for $G^*$; the
reported $G^*$ is the larger of this maximum and the values $G(\rho_j)$ at the normalised elements of the best partition found.

\emph{Optimiser.} \texttt{optimize.extremize} runs Adam with the learning rate $0.05\cdot0.6^r$ in rounds $r=0,\dots,7$, re-centring
the unitary chart ($W_0\leftarrow W$, $X\leftarrow0$) after every round, and returns per start the best value attained at any
iterate; every iterate is an admissible point. All starts for all unitaries form one batch. We used $\CSSteps$ steps for
$n=3,4$ and $\CSFiveSteps$ for $n=5$, and the numbers of starts per unitary given in Table~\ref{tab:cs} ($16$ for the states).
The unitaries $U$ are Haar-random, from the library's sampler (QR decomposition with phase correction~\cite{mezzadri2007}), with
fixed seeds: $\CSThreeU$ for $n=3$, $\CSFourU$ for $n=4$ and $\CSFiveU$ for $n=5$.

\emph{Extended basis search.} A unitary is flagged if the best partition beats the best basis by more than $\CSFlag$. For every
flagged unitary the basis search is repeated with $\CSExtStarts$ further starts, half of them Haar-random and half at bases
built from the directions of the best partition (the principal eigenvectors of its elements, $n$ of them drawn with
probabilities proportional to their traces, perturbed by complex Gaussian noise of size $10^{-3}$ and orthonormalised by the
polar decomposition). For the unitary with the largest remaining gap (for each $n$) another $\CSThreeStressStarts$ starts are run,
again half of each kind.

\emph{Recomputation in high precision.} For every unitary that remains flagged, the best partition and the best basis are
recomputed in mpmath with $\CSDigits$ digits: $U$ is replaced by its polar unitarisation $U'=U(U^*U)^{-1/2}$, exactly unitary at
this precision, and the partition $x_j=S^{-1/2}Z_jZ_j^*S^{-1/2}$ is re-normalised from the stored $Z_j$ (for the basis, $Z_j=w_j$),
so that $|\sum_jx_j-1|$ is of the order $10^{-50}$. The value at such an explicit point is a lower bound for
$H(\Delta_n|U'\Delta_nU'^*)$, up to rounding errors far below the digits we report. This is a high-precision evaluation, not
interval arithmetic; and the basis value is only the best one found, not an upper bound over all bases.

\emph{Structure of a partition.} Elements of trace below $10^{-7}$ times the largest trace are discarded; an element counts as rank
one if its second eigenvalue is at most $10^{-6}$ times its trace; the principal eigenvectors $d,d'$ of two elements are merged
into one direction if $|\langle d,d'\rangle|>1-10^{-4}$. By Lemma~\ref{lem:F}(a), merging exactly parallel rank-one elements does
not change the value.

\subsection{Checks}

For $n=2$ the batched search over general partitions with four elements, for $U=D_1R_\theta D_2$ with random phases, reproduces
Theorem~\ref{thm:n2} to $\CSTwoDevLib$, and the optimal basis of Theorem~\ref{thm:n2}(b), evaluated in NumPy~\cite{harris2020},
gives the closed form to $\CSTwoDevNumpy$ (Table~\ref{tab:cs2}, Figure~\ref{fig:cs}). The test suite compares the closed form
also with a plain BFGS optimisation (SciPy~\cite{virtanen2020}) over partitions of $M_2$ with three elements for four values
of $c$, checks that $300$ random partitions with two to five elements never exceed it, that the value does not depend on the
phases of $U$, the integral representation of $\varphi$ by quadrature, the Bloch reduction of Lemma~\ref{lem:bloch}, the ensemble
identity of Proposition~\ref{prop:psw}(a) for $n=2,3,4$, and the agreement of the mpmath recomputation with the double-precision
value. For complementary masas (Fourier matrices, $n=2,3,4$) the search gives $H=\log n$ to $\CSFourierDev$
(Remark~\ref{rem:envelope}(iii)), and $H(\Delta_3|\Delta_3)$ vanishes to $\CSSelf$.

Table~\ref{tab:cs} and Figure~\ref{fig:cs} summarise the computation for $n=3,4,5$.

\begin{enumerate}
\item[(i)] \emph{Partitions and bases mostly agree.} For most unitaries the best general partition and the best basis agree to
$\CSThreeAgree$, $\CSFourAgree$ and $\CSFiveAgree$ for $n=3,4,5$; for $n=2$ this is Theorem~\ref{thm:n2}.
\item[(ii)] \emph{But not always.} For $\CSThreeFlagged$ of the $\CSThreeU$ unitaries with $n=3$, $\CSFourFlagged$ of the
$\CSFourU$ with $n=4$ and $\CSFiveFlagged$ of the $\CSFiveU$ with $n=5$, a partition beats the best basis found by more than
$\CSFlag$. None of them was cleared by the extended basis search, nor by the further $\CSThreeStressStarts$ starts for the unitary
with the largest gap. The recomputed gaps range from $\CSThreeGapMin$ to $\CSThreeGapMax$ ($n=3$), from $\CSFourGapMin$ to
$\CSFourGapMax$ ($n=4$) and from $\CSFiveGapMin$ to $\CSFiveGapMax$ ($n=5$). For the unitary with the largest gap, the values at the two explicit
points are
\[
  \begin{array}{ccc}
  n & \text{partition: lower bound for }H(\Delta_n|U'\Delta_nU'^*) & \text{best basis found}\\[2pt]
  3 & \CSThreeGeneral & \CSThreeBasis\\
  4 & \CSFourGeneral & \CSFourBasis\\
  5 & \CSFiveGeneral & \CSFiveBasis
  \end{array}
\]
The flagged unitaries are stored in the file
\texttt{connes\_\allowbreak stormer.json} written by the program.
\item[(iii)] \emph{Structure.} In all these cases every element of the best partition has rank one, and the elements point in
$\CSThreeDirs$, $\CSFourDirs$ and $\CSFiveDirs$ distinct directions for $n=3,4,5$, i.e.\ $n+1$; by Lemma~\ref{lem:F}(a) the
partition is, up to the merging tolerance, a partition with $n+1$ rank-one elements. Two of the directions are far from
orthogonal: the largest overlap $|\langle d,d'\rangle|$ is $\CSThreeOverlap$, $\CSFourOverlap$ and $\CSFiveOverlap$ for the
unitaries with the largest gap.
\item[(iv)] \emph{Choda's partition is never optimal.} In every sample the best value found exceeds $h(U)$, by at least
$0.0002$ ($n=3$; Table~\ref{tab:cs}), as for $n=2$ (Corollary~\ref{cor:choda}). Since this value is attained at an explicit
partition, $H(\Delta_n|U\Delta_nU^*)>h(U)$ for every sampled unitary, up to rounding errors. The right panel of
Figure~\ref{fig:cs} shows $H$ against $h(U)$.
\item[(v)] \emph{The bound $G^*$ is not attained.} For every sampled unitary the computed $G^*$ exceeds the computed $H$, by at
least $\CSThreeNaiveMin$, $\CSFourNaiveMin$ and $\CSFiveNaiveMin$ for $n=3,4,5$. Since the computed $G^*$ is attained at an
explicit state and the computed $H$ is only a lower bound for $H$, this is an observation as well: unlike for $n=2$, the optimal
ensembles do not appear to consist of maximisers of $G$.
\end{enumerate}

\begin{observation}\label{obs:bases}
For $n=3,4,5$ and a small but non-negligible fraction of Haar-random unitaries $U$, the supremum~\eqref{eq:cs} for
$H(\Delta_n|U\Delta_nU^*)$ is not attained at an orthonormal basis: partitions with $n+1$ rank-one elements are better than
every basis found by an extensive multistart search.
\end{observation}

\begin{conjecture}\label{conj:n3}
Let $n\ge3$.
\begin{enumerate}
\item[(a)] There are unitaries $U\in U(n)$ (in fact, a set of positive Haar measure) with
$H(\Delta_n|U\Delta_nU^*)>\max_{W\in U(n)}[h(U^*W)-h(W)]$.
\item[(b)] For Haar-almost every $U$, $h(U)<H(\Delta_n|U\Delta_nU^*)<G^*$.
\end{enumerate}
\end{conjecture}

\begin{remark}[Non-abelian subalgebras]\label{rem:nonab}
The same code handles any subalgebra given by an inclusion matrix. For $P=M_2\oplus\C\subset M_3$ and $P=M_2\otimes1\subset M_4$
against $uPu^*$ for two seeded Haar-random $u$ each, the maxima over partitions with $m=n$, $2n$ and $n^2$ elements agree to
$\CSNonabASpread$ and $\CSNonabBSpread$; we obtain $H=\CSNonabA$ and $H=\CSNonabB$, against $G^*=\CSNonabAG$ and
$G^*=\CSNonabBG$. For $M_2\otimes1\subset M_4$, $E_P\rho=\Tr_2\rho\otimes\frac12$ with the partial trace over the second factor, so
$S(E_P\rho)=S(\Tr_2\rho)+\log2\ge\log2$, and similarly $S(E_Q\rho)\le2\log2$; hence $G^*\le\log2$, a bound that the computed $G^*$
reaches while the computed $H$ stays below it. These values are lower bounds found by the search; we have no closed form.
\end{remark}

\section{Open problems}\label{sec:open}

\begin{enumerate}
\item[(1)] \emph{A certificate for Observation~\ref{obs:bases}.} Prove Conjecture~\ref{conj:n3}(a) for one explicit unitary,
e.g.\ the stored $3\times3$ unitary with the largest gap, by a rigorous upper bound for $\max_W[h(U^*W)-h(W)]$ over $U(3)$
(a global optimisation over an eight-dimensional manifold modulo phases, amenable to interval branch and bound).
\item[(2)] \emph{Closed forms for $n\ge3$.} Is there a formula for $H(\Delta_n|U\Delta_nU^*)$ for special families, e.g.\
$U=R_\theta\oplus1_{n-2}$, unitaries close to permutation matrices, or $U$ close to a complementary one? Is $H$ continuous
and piecewise analytic in $U$, with the switch between bases and partitions with $n+1$ elements as a transition?
\item[(3)] \emph{Small angles for $n\ge3$.} For $n=2$, $H$ is of first order in the angle $\theta$, while $h(U)$ is
$O(\theta^2\log\frac1\theta)$ (Theorem~\ref{thm:bound}). Is $H(\Delta_n|U\Delta_nU^*)$ of first order in the distance of $U$ from the
monomial unitaries for every $n$, and with which constant?
\item[(4)] \emph{The number of elements.} Is the supremum for two masas always attained by a partition with at most $n+1$
elements of rank one? Where partitions beat bases, the best partitions found have rank-one elements in $n+1$ directions,
and otherwise a basis ($n$ elements) is as good as the best partition found; Proposition~\ref{prop:psw}(b) only guarantees
$n^2$ elements.
\item[(5)] \emph{The gap to $G^*$.} Characterise the pairs $P,Q$ with $H(P|Q)=G^*$ (Proposition~\ref{prop:envelope}); for masas,
is equality restricted to $n=2$ and to complementary or commuting pairs?
\item[(6)] \emph{Spin model subfactors.} Is the bound of Remark~\ref{rem:bakshi} for $H(R_u|R_v)$ sharp, i.e.\ does
$H(R_u|R_v)=H(\Delta_2|u^*v\Delta_2v^*u)$ hold?
\item[(7)] \emph{Non-abelian subalgebras.} Closed forms for the examples of Remark~\ref{rem:nonab}, starting with
$M_2\otimes1\subset M_4$.
\end{enumerate}
